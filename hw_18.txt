import pandas as pd
import plotly.express as px


months = ['Январь', 'Февраль', 'Март', 'Апрель', 'Май', 'Июнь', 'Июль', 'Август', 'Сентябрь', 'Октябрь', 'Ноябрь', 'Декабрь']

data = {
    'Месяц': months * 2,
    'Год': [2024] * 12 + [2025] * 12,
    'Температура': [
        -15, -25, -15, -5, 7, 20, 30, 30, 12, 2, -10, -15,  # Данные за 2024 год
        -17, -22, -15, -2, 10, 22, 28, 30, 15, 4, -7, -20   # Данные за 2025 год
    ]
}

df = pd.DataFrame(data)

fig = px.bar(df, x='Месяц', y='Температура', animation_frame='Год', title='Изменение температуры в городе по месяцам',
    color='Температура', color_continuous_scale='RdBu')

fig.show()

from dash import Dash, html, dcc
import pandas as pd
import plotly.express as px


x = [1, 2, 3, 4, 5]
y = [1, 4, 9, 16, 25]

df = pd.DataFrame({'x': x, 'y': y})

app = Dash()

fig = px.line(df, x='x', y='y', title='График функции y = x²', markers=True)

app.layout = html.Div([
    html.H1(children=["Мой первый дашборд"]),
    dcc.Graph(figure=fig)
])

if __name__ == '__main__':
    app.run(debug=True)
